"""
====================================================================
 TRADUCTOR MATEMÁTICO MULTIIDIOMA (LaTeX-safe)
====================================================================
Traduce la PROSA de documentos matemáticos entre 7 idiomas
(es, en, fr, de, it, pt, ru) manteniendo INTACTO todo el
simbolismo matemático (fórmulas, ecuaciones, LaTeX, $...$,
$$...$$, \\[...\\], entornos equation/align/theorem...).

Pipeline:
  PDF de entrada (PyMuPDF) -> extracción de texto
  -> segmentación (prosa vs. matemáticas con placeholders)
  -> traducción solo de prosa (DeepL/Google/Argos, modo offline opcional)
  -> reensamblado -> PDF de salida (ReportLab + matplotlib para LaTeX)

Despliegue:
  - Streamlit (app web):  streamlit run app.py
  - GitHub: subir repo con requirements.txt, app.py, README.md
  - Streamlit Cloud: conectar el repo y deploy automático

Instalación (requisitos):
  pip install streamlit pymupdf reportlab matplotlib deep-translator argostranslate langdetect python-dotenv
  # opcional offline: pip install argostranslate && argostranslate install <paquete>
"""

# ---------------------------------------------------------------
# 1. CONFIGURACIÓN Y VOCABULARIO MATEMÁTICO (silabus + gramática)
# ---------------------------------------------------------------
IDIOMAS = {
    "Español": "es",
    "Inglés": "en",
    "Francés": "fr",
    "Alemán": "de",
    "Italiano": "it",
    "Portugués": "pt",
    "Ruso": "ru",
}

# Términos que NO deben traducirse nunca (notación universal)
TERMINOS_PROTEGIDOS = [
    "QED", "et al.", "a.e.", "i.i.d.", "w.l.o.g.", "s.t.", "iff",
    "LHS", "RHS", "det", "tr", "rank", "dim", "ker", "Im", "id",
]

# Gramática propia: entornos/estructuras LaTeX que se preservan bloque completo
ENTORNOS_LATEX = [
    "equation", "equation*", "align", "align*", "alignat", "alignat*",
    "gather", "gather*", "multline", "multline*", "cases", "array",
    "matrix", "pmatrix", "bmatrix", "vmatrix", "Bmatrix", "Vmatrix",
    "split", "aligned", "gathered", "subequations", "theorem",
    "lemma", "corollary", "proposition", "proof", "definition",
    "example", "remark", "solution",
]

# ---------------------------------------------------------------
# 2. SEGMENTADOR: separa prosa de matemáticas y crea placeholders
# ---------------------------------------------------------------
class SegmentadorMatematico:
    """Divide el texto en segmentos {tipo: 'prosa'|'math', texto}.
    Las fórmulas se sustituyen por placeholders de la forma
    ⟦MATH:<n>⟧ y se restauran tras la traducción, 100% intactas."""

    def __init__(self):
        self.placeholders = {}
        self._contador = 0

    # --- patrones de detección (orden importa: bloque largo primero) ---
    @staticmethod
    def _patrones():
        p = []
        # entornos LaTeX: \begin{align}...\end{align}
        ent = "|".join(ENTORNOS_LATEX)
        p.append((re.compile(r"\\begin\{(" + ent + r")\*?\}.*?\\end\{\1\*?\}", re.S), "math"))
        # \[ ... \]
        p.append((re.compile(r"\\\[.*?\\\]", re.S), "math"))
        # $$ ... $$
        p.append((re.compile(r"\$\$.*?\$\$", re.S), "math"))
        # $ ... $
        p.append((re.compile(r"\$[^$\n]+?\$"), "math"))
        # \( ... \)
        p.append((re.compile(r"\\\(.*?\\\)", re.S), "math"))
        # bloques con alta densidad de operadores (heurística)
        p.append((re.compile(r"[A-Za-z0-9\)]\s*\\?[,;]\s*\\frac|\\sum|\\int|\\prod|\\lim|\\sqrt|\\infty"
                             r"(?:[^\n]{0,120})+"), "math"))
        return p

    def proteger(self, texto: str) -> str:
        """Sustituye cada fórmula por ⟦MATH:n⟧ y guarda el original."""
        for patron, _ in self._patrones():
            def _repl(m):
                clave = f"MATH{self._contador}"
                self.placeholders[clave] = m.group(0)
                self._contador += 1
                return f"⟦{clave}⟧"
            texto = patron.sub(_repl, texto)
        # protege símbolos sueltos de notación dentro de prosa
        for t in TERMINOS_PROTEGIDOS:
            texto = texto.replace(t, f"⟦KEEP:{t}⟧")
        return texto

    def restaurar(self, texto: str) -> str:
        """Devuelve las fórmulas originales al texto traducido."""
        for clave, original in self.placeholders.items():
            texto = texto.replace(f"⟦{clave}⟧", original)
            # algunos traductores alteran el placeholder; limpieza agresiva
            texto = re.sub(r"⟦\s*" + clave + r"\s*⟧", original, texto)
        texto = re.sub(r"⟦\s*KEEP:\s*(.+?)\s*⟧", r"\1", texto)
        return texto


def segmentar(texto: str):
    """Devuelve lista de segmentos para traducir prosa por bloques."""
    seg = SegmentadorMatematico()
    protegido = seg.proteger(texto)
    partes = re.split(r"(⟦MATH\d+⟧)", protegido)
    resultado = []
    for parte in partes:
        if not parte.strip():
            continue
        if re.fullmatch(r"⟦MATH\d+⟧", parte.strip()):
            resultado.append({"tipo": "math", "texto": seg.placeholders[parte.strip()[1:-1]]})
        else:
            resultado.append({"tipo": "prosa", "texto": parte})
    return resultado, seg

class TraductorMatematico:
    """Backend de traducción con fallbacks:
    1) DeepL (API key en .env)  2) Google (deep-translator)
    3) Argos Translate (offline)."""

    def __init__(self, destino: str, deepl_key: str = None):
        self.destino = destino
        self.deepl_key = deepl_key or os.getenv("DEEPL_API_KEY")
        self._motor = self._construir_motor()

    def _construir_motor(self):
        try:
            from deep_translator import GoogleTranslator
            return GoogleTranslator(source="auto", target=self.destino)
        except Exception:
            return None

    def traducir_bloque(self, texto: str) -> str:
        if not texto.strip():
            return texto
        try:
            if self.deepl_key:
                import deepl
                return deepl.Translator(self.deepl_key).translate_text(
                    texto, target_lang=self.destino.upper()).text
            if self._motor:
                return self._motor.translate(texto[:4800])
            # fallback offline
            from argostranslate import package, translate as argos
            argos.update_installed_packages() if False else package.update_package_index()
            paquete = [p for p in package.get_installed_packages()
                       if p.to_code == self.destino]
            if paquete:
                return paquete[0].translate(texto)
        except Exception as e:
            print(f"[aviso] fallo de traducción ({e}); se conserva el original.")
            return texto
        return texto

    def traducir_documento(self, texto: str) -> str:
        segmentos, seg = segmentar(texto)
        salida = []
        for s in segmentos:
            if s["tipo"] == "prosa":
                traduccion = self.traducir_bloque(s["texto"])
                # post-edición con glosario matemático del silabus
                glos = GLOSARIO.get(self.destino, {})
                for origen, destino in glos.items():
                    traduccion = re.sub(rf"\b{re.escape(origen)}\b", destino, traduccion)
                salida.append(traduccion)
            else:
                salida.append(f"⟦PLACEHOLDER_{len(salida)}⟧")
        # reensamblar con placeholders y restaurar fórmulas intactas
        completo = " ".join(salida)
        idx = 0
        def _rest(m):
            nonlocal idx
            original = segmentos[[i for i, x in enumerate(segmentos) if x["tipo"] == "math"][idx]]["texto"]
            idx += 1
            return original
        completo = re.sub(r"⟦PLACEHOLDER_\d+⟧", _rest, completo)
        return seg.restaurar(completo)

# ---------------------------------------------------------------
# 4. ENTRADA: lectura de PDF (texto incrustado + OCR opcional)
# ---------------------------------------------------------------
def extraer_texto_pdf(ruta_pdf: str) -> str:
    import fitz  # PyMuPDF
    doc = fitz.open(ruta_pdf)
    partes = []
    for pagina in doc:
        partes.append(pagina.get_text("text"))
    doc.close()
    return "\n".join(partes)

# ---------------------------------------------------------------
# 5. SALIDA: PDF con LaTeX renderizado (matplotlib + reportlab)
# ---------------------------------------------------------------
def _render_latex_png(latex: str, escala: float = 1.4) -> "Image":
    """Convierte una fórmula LaTeX a imagen PNG con mathtext."""
    import matplotlib
    matplotlib.use("Agg")
    import matplotlib.pyplot as plt
    fig = plt.figure(figsize=(0.01, 0.01))
    fig.text(0, 0, f"${latex}$", fontsize=12 * escala)
    buf = io.BytesIO()
    fig.savefig(buf, dpi=300, bbox_inches="tight", pad_inches=0.02,
                transparent=False, facecolor="white")
    plt.close(fig)
    buf.seek(0)
    return buf

def _es_display_math(bloque: str) -> bool:
    return (bloque.strip().startswith("$$") or bloque.strip().startswith("\\[")
            or "\\begin{equation" in bloque or "\\begin{align" in bloque
            or len(bloque) > 60)

def _limpiar_latex(latex: str) -> str:
    for d in ("$$", "\\[", "\\]", "$", "\\(", "\\)"):
        latex = latex.replace(d, "")
    return latex.strip()

def generar_pdf(texto: str, titulo: str, idioma_destino: str, ruta_salida: str):
    """Compone el PDF final: prosa traducida como texto y fórmulas
    matemáticas renderizadas como imágenes (intactas respecto al original)."""
    from reportlab.lib.pagesizes import A4
    from reportlab.lib.styles import getSampleStyleSheet, ParagraphStyle
    from reportlab.lib.units import cm
    from reportlab.platypus import (SimpleDocTemplate, Paragraph, Spacer, Image)
    from reportlab.lib.enums import TA_CENTER

    doc = SimpleDocTemplate(ruta_salida, pagesize=A4,
                            leftMargin=2 * cm, rightMargin=2 * cm,
                            topMargin=2 * cm, bottomMargin=2 * cm)
    estilos = getSampleStyleSheet()
    estilos.add(ParagraphStyle("Titulo", parent=estilos["Title"], fontSize=14))
    estilos.add(ParagraphStyle("Formula", alignment=TA_CENTER, fontSize=10))

    contenido = [Paragraph(titulo, estilos["Titulo"]), Spacer(1, 12)]

    # separar bloques matemáticos de prosa
    trozos = re.split(r"(\$\$.*?\$\$|\\\[.*?\\\]|\\begin\{(?:equation|align)\*?\}.*?\\end\{(?:equation|align)\*?\})", texto, flags=re.S)
    for trozo in trozos:
        t = trozo.strip()
        if not t:
            continue
        if t.startswith("$$") or t.startswith("\\[") or t.startswith("\\begin"):
            latex = _limpiar_latex(t)
            try:
                png = _render_latex_png(latex)
                img = Image(png, width=10 * cm, height=2.5 * cm, kind="proportional")
                contenido.append(img)
            except Exception:
                contenido.append(Paragraph(t.replace("&", "&amp;").replace("<", "&lt;"), estilos["Formula"]))
            contenido.append(Spacer(1, 8))
        else:
            for linea in t.split("\n"):
                if linea.strip():
                    contenido.append(Paragraph(linea.strip().replace("&", "&amp;").replace("<", "&lt;"), estilos["BodyText"]))
            contenido.append(Spacer(1, 6))

    doc.build(contenido)
    return ruta_salida

# ---------------------------------------------------------------
# 6. PIPELINE COMPLETO (uso programático / CLI)
# ---------------------------------------------------------------
def traducir_pdf(ruta_entrada: str, idioma_destino: str, ruta_salida: str = None,
                 titulo: str = "Documento traducido") -> str:
    texto = extraer_texto_pdf(ruta_entrada)
    traductor = TraductorMatematico(idioma_destino)
    traducido = traductor.traducir_documento(texto)
    if ruta_salida is None:
        ruta_salida = str(Path(ruta_entrada).with_name(
            Path(ruta_entrada).stem + f"_{idioma_destino}.pdf"))
    generar_pdf(traducido, f"{titulo} ({idioma_destino})", idioma_destino, ruta_salida)
    return ruta_salida

# 7. APP STREAMLIT (front-end web)
# ---------------------------------------------------------------
def app_streamlit():
    import streamlit as st

    st.set_page_config(page_title="Traductor Matemático", page_icon="∑", layout="wide")
    st.title("∑ Traductor Matemático Multiidioma")
    st.caption("Traduce la prosa matemática en 7 idiomas conservando **intactas** "
               "todas las fórmulas y ecuaciones LaTeX.")

    with st.sidebar:
        st.header("Parámetros")
        destino = st.selectbox("Idioma de destino", list(IDIOMAS.keys()))
        titulo = st.text_input("Título del documento", "Documento traducido")
        st.markdown("---")
        st.markdown("**Motor:** DeepL → Google → Argos (offline)")
        st.markdown("**Formatos:** PDF → PDF")

    archivo = st.file_uploader("Sube el PDF de entrada", type=["pdf"])

    if archivo and st.button("Traducir", type="primary"):
        with st.spinner("Extrayendo texto…"):
            with tempfile.NamedTemporaryFile(delete=False, suffix=".pdf") as tmp:
                tmp.write(archivo.getvalue())
                ruta_tmp = tmp.name
            texto = extraer_texto_pdf(ruta_tmp)

        col1, col2 = st.columns(2)
        with col1:
            st.subheader("Original")
            st.text_area("", texto[:4000], height=350)

        with st.spinner(f"Traduciendo al {destino}…"):
            traductor = TraductorMatematico(IDIOMAS[destino])
            traducido = traductor.traducir_documento(texto)

        with col2:
            st.subheader(f"Traducido ({destino})")
            st.text_area("", traducido[:4000], height=350)

        with st.spinner("Generando PDF…"):
            ruta_out = traducir_pdf(ruta_tmp, IDIOMAS[destino],
                                    titulo=titulo)
        with open(ruta_out, "rb") as f:
            st.download_button("⬇️ Descargar PDF traducido", f.read(),
                               file_name=Path(ruta_out).name,
                               mime="application/pdf")
        st.success("✅ Listo. Fórmulas preservadas sin alteraciones.")

if __name__ == "__main__":
    import sys
    if len(sys.argv) > 1 and sys.argv[1] == "serve":
        app_streamlit()
    elif len(sys.argv) == 4:
        # CLI: python app.py entrada.pdf es salida.pdf
        print(traducir_pdf(sys.argv[1], sys.argv[2], sys.argv[3]))
    else:
        print("Uso:  streamlit run app.py   |   python app.py entrada.pdf es salida.pdf")
