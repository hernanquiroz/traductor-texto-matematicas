# -*- coding: utf-8 -*-
"""OCR para PDFs escaneados usando PyMuPDF + Tesseract.
Requiere el binario de Tesseract instalado en el sistema:
  - Ubuntu/Debian:  sudo apt install tesseract-ocr tesseract-ocr-spa tesseract-ocr-fra tesseract-ocr-deu tesseract-ocr-ita tesseract-ocr-por tesseract-ocr-rus
  - Windows:       https://github.com/UB-Mannheim/tesseract/wiki
  - macOS:          brew install tesseract tesseract-lang
"""
import io
import fitz  # PyMuPDF
import pytesseract
from PIL import Image

# códigos Tesseract por idioma ISO
TESSERACT_LANG = {
    "es": "spa", "en": "eng", "fr": "fra", "de": "deu",
    "it": "ita", "pt": "por", "ru": "rus",
}

def extraer_texto_pdf_ocr(ruta_pdf: str, idiomas=("spa", "eng")) -> str:
    """Rasteriza cada página y aplica OCR. `idiomas` son códigos Tesseract."""
    doc = fitz.open(ruta_pdf)
    partes = []
    for pagina in doc:
        pix = pagina.get_pixmap(dpi=300)          # alta resolución para fórmulas
        img = Image.open(io.BytesIO(pix.tobytes("png")))
        texto = pytesseract.image_to_string(img, lang="+".join(idiomas))
        partes.append(texto)
    doc.close()
    return "\n".join(partes)

def extraer_texto_pdf_hibrido(ruta_pdf: str, idiomas=("spa", "eng")) -> str:
    """Primero intenta texto incrustado; si la página está vacía, hace OCR."""
    doc = fitz.open(ruta_pdf)
    partes, pendientes = [], []
    for i, pagina in enumerate(doc):
        texto = pagina.get_text("text").strip()
        if texto:
            partes.append(texto)
        else:
            pendientes.append(i)
    doc.close()
    if pendientes:  # hay páginas escaneadas
        try:
            ocr = extraer_texto_pdf_ocr(ruta_pdf, idiomas)
            ocr_paginas = ocr.split("\x0c")  # Tesseract separa por form-feed
            for i in pendientes:
                partes.insert(i, ocr_paginas[i] if i < len(ocr_paginas) else "")
        except Exception as e:
            print(f"[aviso] OCR no disponible: {e}")
    return "\n".join(partes)
